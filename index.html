function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; if (data.action === 'history') { var values = sheet.getDataRange().getValues(); var rows = values.slice(1); // ข้าม header var filtered = rows .filter(function(r){ return String(r[0]) === data.name; }) .map(function(r){ return { name: r[0], date: formatValue(r[1], 'dd/MM/yyyy'), time: formatValue(r[2], 'HH:mm:ss'), sys: r[3], dia: r[4], pulse: r[5] }; }); return ContentService.createTextOutput(JSON.stringify({success: true, entries: filtered})) .setMimeType(ContentService.MimeType.JSON); } var targetRow = sheet.getLastRow() + 1; // บังคับคอลัมน์ B (วันที่) และ C (เวลา) ให้เป็น Plain Text ก่อนใส่ค่า // เพื่อไม่ให้ Sheets แปลงเป็น Date แล้วคิดเขตเวลาผิด sheet.getRange(targetRow, 2, 1, 2).setNumberFormat('@'); sheet.getRange(targetRow, 1, 1, 6).setValues( [[data.name, data.date, data.time, data.sys, data.dia, data.pulse]] ); return ContentService.createTextOutput(JSON.stringify({success: true})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({success: false, error: err.message})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function formatValue(value, pattern) { if (Object.prototype.toString.call(value) === '[object Date]') { return Utilities.formatDate(value, 'Asia/Bangkok', pattern); } return value; // เป็นข้อความล้วนอยู่แล้ว ใช้ค่าเดิมได้เลย }
